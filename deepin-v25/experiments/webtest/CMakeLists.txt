cmake_minimum_required(VERSION 3.18)
project(webtest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Quick WebEngineQuick)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(webtest main.cpp)
qt_add_resources(webtest "qml" PREFIX "/" FILES main.qml)
target_link_libraries(webtest PRIVATE Qt6::Quick Qt6::WebEngineQuick)
